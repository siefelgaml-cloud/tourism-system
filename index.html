async function downloadSettlementPDF() { // 1. تحديد الحاوية الرئيسية للجدول const element = document.querySelector('.table-container') || document.getElementById('settlementTableArea'); if (!element) { alert('لم يتم العثور على جدول البيانات للتصدير!'); return; } // 2. عمل نسخة مؤقتة لعدم التأثير على واجهة المستخدم const clone = element.cloneNode(true); // 3. إزالة أزرار الإجراءات (تعديل/حذف) كلياً من النسخة // إزالة رأس العمود (th) الخاص بالإجراءات const headers = clone.querySelectorAll('th'); let actionColumnIndex = -1; headers.forEach((th, index) => { if (th.innerText.includes('إجراءات') || th.classList.contains('no-print')) { actionColumnIndex = index; th.remove(); } }); // إزالة الخلايا (td) المقابلة لعمود الإجراءات من كل صف const rows = clone.querySelectorAll('tr'); rows.forEach(row => { const cells = row.querySelectorAll('td'); if (actionColumnIndex !== -1 && cells[actionColumnIndex]) { cells[actionColumnIndex].remove(); } // إزالة الأزرار إن وجدت داخل الصفوف كاحتياط row.querySelectorAll('.btn, button, .no-print').forEach(btn => btn.remove()); }); // 4. ضبط إعدادات المكتبة (html2pdf) const opt = { margin: [8, 8, 8, 8], // الهوامش بالأبعاد (مم) filename: `سجل_تصفيات_الملفات_${new Date().toISOString().slice(0, 10)}.pdf`, image: { type: 'jpeg', quality: 0.98 }, html2canvas: { scale: 2, // مضاعفة الدقة لمنع تشوه النصوص useCORS: true, // السماح بتحميل الخطوط والألوان scrollX: 0, scrollY: 0, windowWidth: 1200 // محاكاة شاشة عريضة لعدم انضغاط الجدول }, jsPDF: { unit: 'mm', format: 'a4', orientation: 'landscape' }, // اتجاه أفقي لمنع القص pagebreak: { mode: ['avoid-all', 'css', 'legacy'] } // منع قطع الصفوف بين الصفحات }; // 5. بدء التصدير try { await html2pdf().set(opt).from(clone).save(); } catch (error) { console.error('حدث خطأ أثناء إنشاء ملف PDF:', error); } }

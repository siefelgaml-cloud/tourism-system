// دالة حساب وتحديث بطاقات الإجمالي function updateCreditTotals(records) { let totalsByCurrency = {}; records.forEach(item => { const curr = item.currency || 'EGP'; const amt = parseFloat(item.amount) || 0; if (!totalsByCurrency[curr]) { totalsByCurrency[curr] = { deposit: 0, deduction: 0 }; } if (item.type === 'deposit') { totalsByCurrency[curr].deposit += amt; } else if (item.type === 'deduction') { totalsByCurrency[curr].deduction += amt; } }); let depText = [], dedText = [], netText = []; Object.keys(totalsByCurrency).forEach(curr => { const dep = totalsByCurrency[curr].deposit; const ded = totalsByCurrency[curr].deduction; const net = dep - ded; if (dep > 0) depText.push(`${dep.toLocaleString()} ${curr}`); if (ded > 0) dedText.push(`${ded.toLocaleString()} ${curr}`); netText.push(`${net.toLocaleString()} ${curr}`); }); document.getElementById('totalDepositCard').innerText = depText.join(' | ') || '0'; document.getElementById('totalDeductionCard').innerText = dedText.join(' | ') || '0'; document.getElementById('netBalanceCard').innerText = netText.join(' | ') || '0'; } // مراقبة بيانات Firebase وتحديث الجدول والملخص const qCredit = query(collection(db, "credit_balances"), orderBy("createdAt", "desc")); onSnapshot(qCredit, (snapshot) => { const tbody = document.getElementById('creditTableBody'); tbody.innerHTML = ''; currentCreditList = []; if (snapshot.empty) { tbody.innerHTML = 'لا توجد حركة أرصدة كريديت مسجلة حتى الآن'; updateCreditTotals([]); return; } let index = 1; snapshot.forEach((docSnap) => { const data = docSnap.data(); data.id = docSnap.id; currentCreditList.push(data); const dateStr = data.createdAt ? new Date(data.createdAt.seconds * 1000).toLocaleDateString('en-GB') : '-'; const typeBadge = data.type === 'deposit' ? 'إيداع (لنا)' : 'خصم (علينا)'; const row = ` ${index++} ${data.entity || '-'} ${data.fileCode || '-'} ${formatDateToDMY(data.arrivalDate)} ${formatDateToDMY(data.departureDate)} ${typeBadge} ${Number(data.amount || 0).toLocaleString()} ${data.currency || 'EGP'} ${data.description || '-'} ${dateStr} `; tbody.innerHTML += row; }); updateCreditTotals(currentCreditList); }); // فلترة الجدول وتحديث الإجماليات حسب نتائج البحث function filterCreditTable() { const input = document.getElementById("searchCredit"); const filter = input.value.toLowerCase(); const table = document.getElementById("creditTable"); const tr = table.getElementsByTagName("tr"); let filteredRecords = []; for (let i = 1; i < tr.length; i++) { let tdEntity = tr[i].getElementsByTagName("td")[1]; let tdFile = tr[i].getElementsByTagName("td")[2]; if (tdEntity || tdFile) { let entityTxt = tdEntity.textContent || tdEntity.innerText; let fileTxt = tdFile.textContent || tdFile.innerText; if (entityTxt.toLowerCase().indexOf(filter) > -1 || fileTxt.toLowerCase().indexOf(filter) > -1) { tr[i].style.display = ""; // إضافة السجل الظاهر لإعادة حساب إجماليات البحث if (currentCreditList[i - 1]) { filteredRecords.push(currentCreditList[i - 1]); } } else { tr[i].style.display = "none"; } } } updateCreditTotals(filteredRecords); }
